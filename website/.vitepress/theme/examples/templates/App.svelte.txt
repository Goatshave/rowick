<script lang="ts">
import {onDestroy} from 'svelte';

import {createExampleModel,type Density,errorId} from './model';
import type {Person} from './people';
import {theme} from './theme';
import './styles/rowick-theme.css';

import './styles/controls.css';

<p class="sample-announcement" role="status">{state.announcement}</p>
